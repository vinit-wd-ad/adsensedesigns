RewriteEngine On

# UNIVERSAL FIX: Base directory is set to root (/) for uniform path resolution
RewriteBase /

# Admin Area Bypass: Exclude all admin dashboard paths from frontend routing rules
RewriteCond %{REQUEST_URI} ^/admin [NC]
RewriteRule .* - [L]

# ====================================================================
# DYNAMIC BLOGS ROUTING ELEMENTS
# ====================================================================
# Routes clean URLs like example.com/blog/slug-name internally to blog-details.php
RewriteRule ^blog/([a-zA-Z0-9_-]+)/?$ blog-details.php?slug=$1 [QSA,L]

# ====================================================================
# DYNAMIC SERVICES ROUTING ELEMENTS
# ====================================================================
# Routes clean URLs like example.com/service-slug internally to service.php
# Executes only if the requested path is not a physical file, directory, or blog route
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_URI} !^/blog/ [NC]
RewriteRule ^([a-zA-Z0-9_-]+)/?$ service.php?slug=$1 [QSA,L]

# ====================================================================
# ANTI-SPAM AND INVALID REQUESTS MANAGEMENT
# ====================================================================
# Blocks direct external query parameters string access to index or specific target landing files
RewriteCond %{THE_REQUEST} ^[A-Z]{3,}\s/(why-us|index)(\.php)?\?[^\s]*\sHTTP/
RewriteRule ^(why-us|index)(\.php)?$ invalid-page [R=301,L]

# Redirects obsolete dynamic numeric HTML requests safely to invalid-page placeholder
RewriteCond %{REQUEST_URI} ^/([0-9]+)(\.htm)?$ [NC]
RewriteRule ^ invalid-page [R=301,L]

# Global blocklist mapping rules targeting common vulnerability scanner directory hits
RewriteRule ^(shopdetail|oreka|toreka|feature|pcmypage|product_archives|snoox|safe_search|product|zhHant)(/.*)?$ invalid-page [R=301,L]

# ====================================================================
# EXTENSION REMOVAL AND CLEAN URL ARCHITECTURE
# ====================================================================
# Forces internal canonical 301 redirection if a user tries to access a .php file extension explicitly
RewriteCond %{THE_REQUEST} ^[A-Z]{3,}\s/([^.]+)\.php\sHTTP/
RewriteRule ^([^.]+)\.php$ $1 [R=301,L]

# Standard internal routing backup lookup for server files matching extensionless routes
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_URI} !^/blog/ [NC]
RewriteRule ^([a-zA-Z0-9_-]+)$ $1.php [L]

# ====================================================================
# CORE SYSTEM CUSTOM SYSTEM FALLBACKS & RESPONSE HEADERS
# ====================================================================
ErrorDocument 404 /404.php
